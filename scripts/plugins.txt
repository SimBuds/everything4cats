# The plugin baseline, one slug per line.
#
# Single source of truth, read by scripts/provision.sh so the server and the
# local container cannot drift. Blank lines and # comments are ignored.
#
# A trailing ":inactive" means install but do not activate.
#
# Reviewed 2026-08-10 for Everything4Cats. Nothing here is topic-specific: an
# affiliate review site needs the same SEO, link-cloaking, redirect, mail and
# backup layers whatever it writes about.

secure-custom-fields            # the custom fields plugins/e4c-content registers; see the note below
seo-by-rank-math                # SEO, sitemaps, and Article schema
fluent-smtp                     # transactional mail relay; without it password resets fail silently
updraftplus                     # backups, configured to off-site storage before launch
redirection                     # 301s, so reorganising the site does not shed links
limit-login-attempts-reloaded   # brute-force throttling, paired with fail2ban at the OS layer
complianz-gdpr                  # Canadian and GDPR consent
thirstyaffiliates               # affiliate link cloaking and tracking

wp-super-cache:inactive         # activated only after the site is live; caching hides changes
google-site-kit:inactive        # inert until its Google OAuth completes against a real domain

# A trailing ":delete" removes a plugin WordPress ships inside its own tarball.
# These are not installed from wordpress.org, so leaving them out of this list is
# not enough: `wp core download` puts them back on every fresh install.
#
# Added 2026-08-11, after both were deleted by hand on the live server. Keeping
# them here rather than hardcoded in provision.sh is what makes this file the
# single source of truth for plugin state rather than most of it.
akismet:delete                  # comment spam filter; comments are not part of this site's plan
hello:delete                    # Hello Dolly, a 2004 joke plugin that prints song lyrics in the admin

# ---------------------------------------------------------------------------
# secure-custom-fields, and why it is the field layer rather than ACF Pro.
#
# plugins/e4c-content registers its field groups through ACF's API. Four of the
# eight fields are `repeater`, which was a Pro-only field type, so this line
# used to read "CANNOT be in this list, and still required": ACF Pro is
# commercial and not on wordpress.org, so no slug here could install it, and it
# had to be bought and licensed by hand in wp-admin.
#
# Replaced 2026-08-13 with Secure Custom Fields, WordPress.org's October 2024
# fork of ACF. Verified against the published 6.9.5 package before switching,
# rather than assumed: every field type e4c-content uses is present (text,
# textarea, url, post_object, repeater), flexible-content, clone and gallery
# ship too, and acf_add_local_field_group, get_field and have_rows all still
# resolve under their ACF names. No PHP in the theme or in e4c-content changed.
#
# The saving is not really the licence fee. It is that this file can describe
# the whole plugin baseline again, so a rebuilt host needs no manual plugin
# step, and the container harness exercises the real get_field() path instead
# of only the raw post-meta fallback in the theme's e4c_field().
#
# If it is ever absent, nothing breaks loudly, which is what makes it easy to
# miss. Post types still register, reviews stay published, e4c-content prints
# an admin notice, and e4c_field() falls back to raw post meta. What is missing
# is the editing UI.
# ---------------------------------------------------------------------------
